using Bidulgi;
public class PlayerMove : Behaviour
{
[Edit, Range(0, 20)] public float speed = 5.5f;
[Auto] CharacterBody body; // Get<CharacterBody>() 자동
void Update(float dt)
{
Vector2 move = Input.Value<Vector2>(Actions.Move);
body.Move(new Vector3(move.x, 0, move.y) * speed);
if (Input.Pressed(Actions.Jump)) body.Jump(4.5f);
}
async void OnLanded()
{ Vfx.Play(Effects.Dust, transform.position);
await transform.Tween(scale: 0.9f, seconds: 0.06f, pingPong: true);
}
}
