# ---- Reflection: Bidulgi Header Tool (queue 1-2) ----
# bdg_reflect(<target> <header>...): runs tools/bht/bht.py on each header (only when it or the tool changed)
# and adds <stem>.gen.cpp plus <target>_reflect.cpp, which defines bdg_reflect_register_<target>().
function(bdg_reflect target)
  set(decls "")
  set(calls "")
  foreach(header ${ARGN})
    get_filename_component(stem ${header} NAME_WE)
    string(MAKE_C_IDENTIFIER ${stem} id)
    set(out ${CMAKE_BINARY_DIR}/gen/${target}/${stem}.gen.cpp)
    add_custom_command(OUTPUT ${out}
      COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tools/bht/bht.py ${CMAKE_SOURCE_DIR}/${header} ${out} ${header}
      DEPENDS ${CMAKE_SOURCE_DIR}/${header} ${CMAKE_SOURCE_DIR}/tools/bht/bht.py
      COMMENT "bht ${header}" VERBATIM)
    target_sources(${target} PRIVATE ${out})
    string(APPEND decls "void bdg_reflect_register_${id}();\n")
    string(APPEND calls "  bdg_reflect_register_${id}();\n")
  endforeach()
  set(reg ${CMAKE_BINARY_DIR}/gen/${target}/${target}_reflect.cpp)
  file(CONFIGURE OUTPUT ${reg} CONTENT "// Generated by bdg_reflect() in CMakeLists.txt. Do not edit.\n${decls}void bdg_reflect_register_${target}() {\n${calls}}\n")
  target_sources(${target} PRIVATE ${reg})
endfunction()

# ---- C# scripting (queue 4-1): Bidulgi.Scripting.dll into <exe folder>/Managed ----
# The component classes come from reflection: bdg writes Bindings.g.cs, then `dotnet build` compiles the assembly.
# Without the .NET SDK the engine still builds; C# scripting then reports that the assembly is missing.
find_program(BDG_DOTNET dotnet)
if(BDG_DOTNET)
  set(BDG_BINDINGS ${CMAKE_BINARY_DIR}/Generated/Bindings.g.cs)
  file(GLOB BDG_SCRIPTING_SOURCES CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/Scripting/Bidulgi.Scripting/*.cs ${CMAKE_SOURCE_DIR}/Scripting/Bidulgi.Scripting/*.csproj)
  # bdg rewrites Bindings.g.cs only when the text changed, so the C# build reruns only for real API changes
  add_custom_command(OUTPUT ${CMAKE_BINARY_DIR}/Generated/Bindings.stamp BYPRODUCTS ${BDG_BINDINGS}
    COMMAND $<TARGET_FILE:bdg> csharp-bindings ${BDG_BINDINGS}
    COMMAND ${CMAKE_COMMAND} -E touch ${CMAKE_BINARY_DIR}/Generated/Bindings.stamp
    DEPENDS bdg
    COMMENT "C# bindings from reflection")
  set(BDG_MANAGED_DLL ${CMAKE_BINARY_DIR}/$<CONFIG>/Managed/Bidulgi.Scripting.dll)
  add_custom_command(OUTPUT ${BDG_MANAGED_DLL}
    COMMAND ${BDG_DOTNET} build ${CMAKE_SOURCE_DIR}/Scripting/Bidulgi.Scripting/Bidulgi.Scripting.csproj -c $<CONFIG> --nologo -v:q
            -o ${CMAKE_BINARY_DIR}/$<CONFIG>/Managed -p:BdgBindings=${BDG_BINDINGS}
    COMMAND ${CMAKE_COMMAND} -E touch ${BDG_MANAGED_DLL}
    # the stamp (a real OUTPUT) orders this after bdg in a clean build; Visual Studio does not order rules by BYPRODUCTS
    DEPENDS ${CMAKE_BINARY_DIR}/Generated/Bindings.stamp ${BDG_BINDINGS} ${BDG_SCRIPTING_SOURCES}
    COMMENT "Bidulgi.Scripting.dll (dotnet build)")
  # Sample Behaviours (4-2): a game-style script assembly built against the one above (also the lifecycle tests)
  set(BDG_SAMPLES_DLL ${CMAKE_BINARY_DIR}/$<CONFIG>/Managed/Bidulgi.Samples.dll)
  file(GLOB BDG_SAMPLES_SOURCES CONFIGURE_DEPENDS ${CMAKE_SOURCE_DIR}/Scripting/Bidulgi.Samples/*.cs ${CMAKE_SOURCE_DIR}/Scripting/Bidulgi.Samples/*.csproj)
  add_custom_command(OUTPUT ${BDG_SAMPLES_DLL}
    COMMAND ${BDG_DOTNET} build ${CMAKE_SOURCE_DIR}/Scripting/Bidulgi.Samples/Bidulgi.Samples.csproj -c $<CONFIG> --nologo -v:q
            -o ${CMAKE_BINARY_DIR}/$<CONFIG>/Managed -p:BdgManaged=${CMAKE_BINARY_DIR}/$<CONFIG>/Managed
    COMMAND ${CMAKE_COMMAND} -E touch ${BDG_SAMPLES_DLL}
    DEPENDS ${BDG_MANAGED_DLL} ${BDG_SAMPLES_SOURCES}
    COMMENT "Bidulgi.Samples.dll (dotnet build)")
  # The same samples built with HOTRELOAD_V2 (a changed Spinner): the hot reload test swaps it in while playing (4-4)
  set(BDG_SAMPLES_V2_DLL ${CMAKE_BINARY_DIR}/$<CONFIG>/Managed/HotReloadV2/Bidulgi.Samples.dll)
  add_custom_command(OUTPUT ${BDG_SAMPLES_V2_DLL}
    COMMAND ${BDG_DOTNET} build ${CMAKE_SOURCE_DIR}/Scripting/Bidulgi.Samples/Bidulgi.Samples.csproj -c $<CONFIG> --nologo -v:q
            -o ${CMAKE_BINARY_DIR}/$<CONFIG>/Managed/HotReloadV2 -p:BdgManaged=${CMAKE_BINARY_DIR}/$<CONFIG>/Managed
            -p:DefineConstants=HOTRELOAD_V2 -p:IntermediateOutputPath=obj/v2/$<CONFIG>/
    COMMAND ${CMAKE_COMMAND} -E touch ${BDG_SAMPLES_V2_DLL}
    DEPENDS ${BDG_SAMPLES_DLL} ${BDG_SAMPLES_SOURCES}
    COMMENT "Bidulgi.Samples.dll, hot reload variant (dotnet build)")
  add_custom_target(BidulgiScripting ALL DEPENDS ${CMAKE_BINARY_DIR}/Generated/Bindings.stamp ${BDG_MANAGED_DLL} ${BDG_SAMPLES_DLL} ${BDG_SAMPLES_V2_DLL})
  add_dependencies(BidulgiEditor BidulgiScripting)
  add_custom_command(TARGET BidulgiEditor POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_BINARY_DIR}/$<CONFIG>/Managed ${CMAKE_SOURCE_DIR}/Bin/Managed)
else()
  message(WARNING "dotnet not found: C# scripting (Bidulgi.Scripting.dll) is not built")
endif()
